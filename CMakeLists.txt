cmake_minimum_required(VERSION 3.20)
project(x360_reconstruction LANGUAGES CXX)

target_include_directories(x360_execution PUBLIC include)

target_include_directories(x360_machine PUBLIC include)

target_link_libraries(x360_xam_xui PUBLIC x360_formats)
add_executable(x360_resource_probe tools/native/resource_probe.cpp)
target_link_libraries(x360_resource_probe PRIVATE x360_formats x360_xam_xui)
add_executable(x360_module_probe tools/native/module_probe.cpp)
target_link_libraries(x360_module_probe PRIVATE x360_metro_17559)
