cmake_minimum_required(VERSION 3.20)
project(x360_reconstruction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(x360_execution STATIC runtime/execution/dispatch.cpp)
target_include_directories(x360_execution PUBLIC include)
add_library(x360_machine STATIC runtime/machine/address_space.cpp)
target_include_directories(x360_machine PUBLIC include)
add_library(x360_dashboard_faithful STATIC src/os/dashboard/leaf_17559.cpp src/os/dashboard/memory_leaf_17559.cpp)
target_include_directories(x360_dashboard_faithful PUBLIC include)
include(CTest)
if(BUILD_TESTING)
    add_executable(x360_native_tests tests/native/test_dispatch.cpp)
    target_link_libraries(x360_native_tests PRIVATE x360_execution x360_dashboard_faithful)
    add_test(NAME native_leaf_dispatch COMMAND x360_native_tests)
    add_executable(x360_memory_tests tests/native/test_memory.cpp)
    target_link_libraries(x360_memory_tests PRIVATE x360_machine x360_dashboard_faithful)
    add_test(NAME native_memory_leaf COMMAND x360_memory_tests)
endif()
